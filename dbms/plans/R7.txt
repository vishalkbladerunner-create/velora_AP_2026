-- the same list filtered by area, case-insensitively (GET /pharmacies?area=Model Town).
SELECT
    p."id" AS pharmacy_id,
    p."name" AS pharmacy_name,
    p."area",
    p."address",
    p."phone",
    p."licenceNumber",
    p."status",
    s."id" AS stock_id,
    s."quantity",
    s."pricePaise",
    s."createdAt",
    s."updatedAt",
    m."id" AS medicine_id,
    m."name" AS medicine_name,
    m."form"
FROM "Pharmacy" p
LEFT JOIN "StockItem" s ON s."pharmacyId" = p."id"
LEFT JOIN "Medicine" m ON m."id" = s."medicineId"
WHERE p."status" = 'VERIFIED'
  AND p."area" ILIKE '%Model Town%'
ORDER BY p."name" ASC, m."name" ASC;
