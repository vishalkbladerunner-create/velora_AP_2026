-- R9
-- second step of the medicine search — stock rows in verified pharmacies, quantity above zero, cheapest first, with the pharmacy and medicine attached. The area predicate is optional; it is kept here to match GET /pharmacies?medicine=dolo&area=Sector 14. The app binds the ids returned by R8; the subquery below is equivalent when the two steps are run as one statement.
SELECT
    s."id" AS stock_id,
    s."pharmacyId",
    s."medicineId",
    s."quantity",
    s."pricePaise",
    s."createdAt",
    s."updatedAt",
    p."id" AS pharmacy_id,
    p."name" AS pharmacy_name,
    p."area",
    p."address",
    p."phone",
    p."licenceNumber",
    p."status",
    m."id" AS medicine_id,
    m."name" AS medicine_name,
    m."form"
FROM "StockItem" s
JOIN "Pharmacy" p ON p."id" = s."pharmacyId"
JOIN "Medicine" m ON m."id" = s."medicineId"
WHERE s."medicineId" IN (
        SELECT "id"
        FROM "Medicine"
        WHERE "name" ILIKE '%dolo%'
    )
  AND s."quantity" > 0
  AND p."status" = 'VERIFIED'
  AND p."area" ILIKE '%Sector 14%'
ORDER BY s."pricePaise" ASC;
